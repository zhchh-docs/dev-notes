45.1 Vitest 环境与基础

npm i -D vitest @vue/test-utils @pinia/testing jsdom

为什么测试 store 这么简单？因为 store 不依赖 DOM——它就是一个 JS 模块。你只需要在每次测试前给它一个干净的 pinia 实例：

// stores/counter.spec.js
import { setActivePinia, createPinia } from 'pinia'
import { beforeEach, describe, it, expect } from 'vitest'
import { useCounterStore } from './counter'

describe('counterStore', () => {
  // 每个测试前新建一个干净的 pinia
  beforeEach(() => {
    setActivePinia(createPinia())
  })

  it('初始 count 为 0', () => {
    const store = useCounterStore()
    expect(store.count).toBe(0)
  })
})

setActivePinia(createPinia()) 是关键：它告诉 Pinia"当前测试用这个新的 pinia 实例"，避免测试之间互相污染。

45.2 直接测 store

import { setActivePinia, createPinia } from 'pinia'
import { beforeEach, describe, it, expect } from 'vitest'
import { useCounterStore } from './counter'

describe('counterStore', () => {
  beforeEach(() => setActivePinia(createPinia()))

  it('action increment 让 count +1', () => {
    const store = useCounterStore()
    expect(store.count).toBe(0)

    store.increment()
    expect(store.count).toBe(1)
  })

  it('action incrementBy(n) 加 n', () => {
    const store = useCounterStore()
    store.incrementBy(5)
    expect(store.count).toBe(5)
  })

  it('getter double 是 count 的 2 倍', () => {
    const store = useCounterStore()
    store.count = 3
    expect(store.double).toBe(6)
  })

  it('$patch 批量改 state', () => {
    const store = useCounterStore()
    store.$patch({ count: 10, history: ['+10'] })
    expect(store.count).toBe(10)
    expect(store.history).toEqual(['+10'])
  })

  it('$reset 重置回初始值', () => {
    const store = useCounterStore()
    store.increment()
    store.increment()
    store.$reset()
    expect(store.count).toBe(0)
  })
})

测 store 和测普通类没区别：new 出来、调方法、断言属性。

45.3 @pinia/testing

@pinia/testing 提供一个"测试专用的 pinia"，它有一个关键能力：把所有 action 自动 stub 掉（不真正执行），这样组件测试时不用担心 action 发请求。

import { createTestingPinia } from '@pinia/testing'
import { mount } from '@vue/test-utils'
import MyComponent from './MyComponent.vue'
import { useUserStore } from '@/stores/user'

// 创建测试 pinia：默认 stub 所有 action
const pinia = createTestingPinia({
  // 选项（具体字段以官方文档为准）：
  stubActions: true,        // action 不真正执行，只记录调用
  // 也可以给初始 state：
  initialState: {
    user: { name: '测试用户' }
  }
})

mount(MyComponent, {
  global: { plugins: [pinia] }
})

// 之后在测试里可以断言 action 被调过：
const userStore = useUserStore()
expect(userStore.fetchUser).toHaveBeenCalled()
场景用普通 pinia用 testing pinia
测 store 本身✅ 推荐没必要
测组件（不关心 action 实现）要自己 mock✅ 自动 stub
测 store 的异步 action✅ 自己 mock 接口不适用
版本说明：@pinia/testing 的具体选项（stubActions、createSpy、fakeAppId 等）以官方文档为准，不同版本略有差异。Pinia 2.x 配合 @pinia/testing 0.x。

45.4 异步 action 测试

异步 action 通常会发请求。测试时把请求层 mock 掉：

// stores/user.js
import { defineStore } from 'pinia'
import api from '@/api'   // 接口层

export const useUserStore = defineStore('user', {
  state: () => ({ user: null, loading: false }),
  actions: {
    async fetchUser(id) {
      this.loading = true
      try {
        this.user = await api.get(`/users/${id}`)
      } finally {
        this.loading = false
      }
    }
  }
})
// stores/user.spec.js
import { setActivePinia, createPinia } from 'pinia'
import { beforeEach, describe, it, expect, vi } from 'vitest'
import { useUserStore } from './user'
import api from '@/api'

// mock 整个 api 模块
vi.mock('@/api', () => ({
  default: {
    get: vi.fn()   // 返回一个 jest/vitest mock 函数
  }
}))

describe('userStore', () => {
  beforeEach(() => {
    setActivePinia(createPinia())
    vi.clearAllMocks()
  })

  it('fetchUser 成功后设置 user', async () => {
    // 让 api.get 返回测试数据
    api.get.mockResolvedValue({ id: 1, name: '测试用户' })

    const store = useUserStore()
    await store.fetchUser(1)

    expect(api.get).toHaveBeenCalledWith('/users/1')
    expect(store.user).toEqual({ id: 1, name: '测试用户' })
    expect(store.loading).toBe(false)
  })

  it('fetchUser 失败时 loading 变回 false', async () => {
    api.get.mockRejectedValue(new Error('网络错误'))

    const store = useUserStore()
    await expect(store.fetchUser(1)).rejects.toThrow('网络错误')
    expect(store.loading).toBe(false)
  })
})

45.5 组件测试中使用 store

组件测试时，要把 pinia 作为插件传给 mount：

import { mount } from '@vue/test-utils'
import { createPinia, setActivePinia } from 'pinia'
import UserProfile from './UserProfile.vue'
import { useUserStore } from '@/stores/user'

describe('UserProfile', () => {
  it('显示 user.name', async () => {
    const pinia = createPinia()
    setActivePinia(pinia)

    const wrapper = mount(UserProfile, {
      global: {
        plugins: [pinia]    // 关键：把 pinia 装进组件
      }
    })

    // 拿到 store 后改 state，看组件是否更新
    const userStore = useUserStore()
    userStore.name = '测试用户'

    await wrapper.vm.$nextTick()
    expect(wrapper.text()).toContain('测试用户')
  })
})

45.6 本章常见坑

  • 坑 1测试之间共享 store 状态：忘了 beforeEach 里 setActivePinia(createPinia())，上一个测试改的 state 污染下一个。
  • 坑 2异步 action 测试没 await：测试跑完 action 还没结束，断言失败。所有 await 都要写。
  • 坑 3组件测试忘了把 pinia 装进 plugins：组件里 useStore() 报错"no active pinia"。
  • 坑 4用真实接口测试：测试慢、不稳定、依赖网络。永远 mock 接口层。
  • 坑 5测试 action 内部实现细节：比如断言"fetchUser 里先 set loading=true"。应该测外部行为（结果），不测内部步骤。
