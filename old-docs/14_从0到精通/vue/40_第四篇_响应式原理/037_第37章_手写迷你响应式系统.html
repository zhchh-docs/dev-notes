37.1 第一步：track / trigger 骨架

// ========== 1. 依赖表与 activeEffect ==========
const targetMap = new WeakMap()
let activeEffect = null
let shouldTrack = true

class ReactiveEffect {
  constructor(fn, scheduler) {
    this.fn = fn
    this.scheduler = scheduler
    this.deps = []           // 这个 effect 登记过的所有 dep（用于 cleanup）
  }
  run() {
    activeEffect = this
    shouldTrack = true
    try {
      return this.fn()
    } finally {
      activeEffect = null
      shouldTrack = false
    }
  }
}

// ========== 2. track ==========
function track(target, key) {
  if (!shouldTrack || !activeEffect) return

  let depsMap = targetMap.get(target)
  if (!depsMap) targetMap.set(target, (depsMap = new Map()))

  let dep = depsMap.get(key)
  if (!dep) depsMap.set(key, (dep = new Set()))

  dep.add(activeEffect)
  activeEffect.deps.push(dep)
}

// ========== 3. trigger ==========
function trigger(target, key) {
  const depsMap = targetMap.get(target)
  if (!depsMap) return

  const dep = depsMap.get(key)
  if (!dep) return

  // 复制一份，避免 effect.run() 内部 cleanup 又改 Set 引起遍历异常
  const effectsToRun = new Set(dep)
  effectsToRun.forEach(effect => {
    if (effect.scheduler) {
      effect.scheduler(effect)
    } else {
      effect.run()
    }
  })
}

// ========== 4. cleanup（effect 重跑前先删自己）==========
function cleanupEffect(effect) {
  effect.deps.forEach(dep => dep.delete(effect))
  effect.deps.length = 0
}

// ========== 5. effect 入口 ==========
function effect(fn, options = {}) {
  const ef = new ReactiveEffect(fn, options.scheduler)
  ef.run()
  return () => {
    cleanupEffect(ef)
  }
}

37.2 第二步：reactive 与 Proxy

const reactiveMap = new WeakMap()

function isObject(v) { return v !== null && typeof v === 'object' }
function hasChanged(a, b) { return !Object.is(a, b) }

function reactive(target) {
  if (!isObject(target)) return target

  // 命中缓存：同一个原始对象返回同一个代理
  const existing = reactiveMap.get(target)
  if (existing) return existing

  const proxy = new Proxy(target, {
    get(target, key, receiver) {
      if (key === '__v_raw') return target       // 反查原始对象
      const res = Reflect.get(target, key, receiver)
      track(target, key)
      // 惰性：嵌套对象才代理
      return isObject(res) ? reactive(res) : res
    },
    set(target, key, value, receiver) {
      const hadKey = Object.prototype.hasOwnProperty.call(target, key)
      const oldValue = Reflect.get(target, key, receiver)
      Reflect.set(target, key, value, receiver)
      if (!hadKey || hasChanged(value, oldValue)) {
        trigger(target, key)
      }
      return true
    },
    deleteProperty(target, key) {
      const hadKey = Object.prototype.hasOwnProperty.call(target, key)
      const result = Reflect.deleteProperty(target, key)
      if (hadKey) trigger(target, key)
      return result
    }
  })
  reactiveMap.set(target, proxy)
  return proxy
}

37.3 第三步：ref 与 computed

// ========== ref ==========
class RefImpl {
  constructor(value) {
    this.__v_isRef = true
    this._rawValue = value
    this._value = isObject(value) ? reactive(value) : value
  }
  get value() {
    track(this, 'value')
    return this._value
  }
  set value(newVal) {
    if (hasChanged(newVal, this._rawValue)) {
      this._rawValue = newVal
      this._value = isObject(newVal) ? reactive(newVal) : newVal
      trigger(this, 'value')
    }
  }
}

function ref(value) { return new RefImpl(value) }

function isRef(r) { return !!(r && r.__v_isRef) }
function unref(r) { return isRef(r) ? r.value : r }

// ========== computed ==========
class ComputedRefImpl {
  constructor(getter, setter) {
    this._getter = getter
    this._setter = setter
    this._value = undefined
    this.dirty = true

    // 内部 effect：依赖变了只标 dirty + 通知下游
    this.effect = new ReactiveEffect(getter, {
      scheduler: () => {
        if (!this.dirty) {
          this.dirty = true
          trigger(this, 'value')
        }
      }
    })
  }
  get value() {
    track(this, 'value')
    if (this.dirty) {
      this._value = this.effect.run()
      this.dirty = false
    }
    return this._value
  }
  set value(newVal) {
    this._setter && this._setter(newVal)
  }
}

function computed(getterOrOptions) {
  if (typeof getterOrOptions === 'function') {
    return new ComputedRefImpl(getterOrOptions)
  }
  return new ComputedRefImpl(getterOrOptions.get, getterOrOptions.set)
}

37.4 第四步：watchEffect 与 watch

// ========== watchEffect ==========
function watchEffect(source) {
  let cleanup = null
  const onCleanup = (fn) => { cleanup = fn }

  const ef = new ReactiveEffect(function () {
    if (cleanup) cleanup()           // 跑新一次前先跑上次的清理
    source(onCleanup)
  })
  ef.run()

  return () => cleanupEffect(ef)
}

// ========== watch ==========
function watch(source, cb, options = {}) {
  let getter
  if (isRef(source)) {
    getter = () => source.value
  } else if (typeof source === 'function') {
    getter = source
  } else if (isObject(source)) {
    getter = () => source
    options.deep = true
  }

  let oldVal
  let cleanup = null
  const onCleanup = (fn) => { cleanup = fn }

  const ef = new ReactiveEffect(
    () => getter(),
    {
      scheduler: () => {
        // 依赖变了：重跑 getter 拿新值，再调 cb
        if (cleanup) cleanup()
        const newVal = ef.run()
        cb(newVal, oldVal, onCleanup)
        oldVal = newVal
      }
    }
  )

  if (options.immediate) {
    // 立即跑一次 cb（旧值为 undefined）
    oldVal = undefined
    const newVal = ef.run()
    cb(newVal, oldVal, onCleanup)
    oldVal = newVal
  } else {
    // 不立即：先拿一次旧值
    oldVal = ef.run()
  }

  return () => cleanupEffect(ef)
}

37.5 验证用例

// 用例 1：reactive 修改触发
const state = reactive({ count: 0, user: { name: 'vue' } })
effect(() => {
  console.log('effect1: count =', state.count)
})
state.count = 1              // 打印 effect1: count = 1
state.count = 2             // 打印 effect1: count = 2

// 用例 2：ref
const r = ref(0)
effect(() => console.log('effect2: r =', r.value))
r.value = 10                 // 打印 effect2: r = 10

// 用例 3：computed 缓存
const a = ref(1)
const b = computed(() => {
  console.log('computed getter 跑了')
  return a.value * 2
})
console.log('b1 =', b.value)  // 打印 getter + b1 = 2
console.log('b2 =', b.value)  // 不打印 getter，b2 = 2（缓存）
a.value = 5
console.log('b3 =', b.value)  // 打印 getter + b3 = 10

// 用例 4：watch 新旧值
watch(() => state.count, (newVal, oldVal) => {
  console.log('watch: count', oldVal, '->', newVal)
})
state.count = 100            // 打印 watch: count 2 -> 100

// 用例 5：watchEffect 竞态清理
let seq = 0
watchEffect(async (onCleanup) => {
  const id = ++seq
  onCleanup(() => { if (id === seq) console.log('cleanup id', id) })
  const result = await Promise.resolve(id)
  console.log('effect resolve id', result)
})

把以上 5 段按顺序跑一遍，应该看到：

  • 用例 1：effect1 跑 3 次（初始化 + 两次 set）；
  • 用例 2：effect2 跑 2 次；
  • 用例 3：computed getter 跑 2 次（首次 + a 变脏后），中间那次读不跑；
  • 用例 4：watch 打印 2 -> 100；
  • 用例 5：每次新一轮 effect 开始前先跑上次的 cleanup。

37.6 与 Vue 源码的差异

上面这 200 行能跑，但和 Vue 3.4 的真实响应式系统比，简化了不少：

能力我们的迷你版Vue 真实实现
批处理trigger 同步跑queueJob + 微任务 flush，去重 + 排序
数组处理只处理普通 set下标 / length 联动 + 7 个变异方法包装
Map / Set不支持方法重写（get/set/has/...）
deep 遍历不支持traverse 递归读所有嵌套字段
调度器effect 级 schedulerpre/post/sync 三队列 + 组件生命周期顺序
ref 解包只支持 ref() 本身reactive 对象字段自动解包、模板编译期解包
markRaw / shallow不支持反查缓存 + 跳过代理
性能优化无pauseTracking、dep 双向 cleanup、effect 栈

为什么"这样设计"——第四篇总结

  • 为什么 Proxy + Reflect：能力完整（新增属性、数组下标、枚举），Reflect 修正 receiver；
  • 为什么 track 用三层 Map：WeakMap 防泄漏、Map 装属性、Set 自动去重；
  • 为什么 activeEffect 是全局游标：get 拦截点本身不知道"谁在读"，靠这一瞬的全局变量建立关联；
  • 为什么 computed 要 lazy + dirty：不被读不计算、依赖没变不重算；
  • 为什么更新要批处理：避免同步代码里 N 次触发 N 次渲染。

Vue 响应式的所有"奇技淫巧"，归结起来就一句话：读的时候记一笔（track），写的时候按名单通知（trigger），通知时尽量批处理、尽量缓存。
