67.1 每日 GMV 与环比

环比 = 今天 vs 昨天，用 LAG 一次算出来，不用自 JOIN。

-- 每日 GMV
WITH daily AS (
  SELECT DATE(payed_at) AS dt, SUM(total_amount) AS gmv, COUNT(*) AS orders
  FROM orders
  WHERE status = 'PAYED' AND payed_at IS NOT NULL
  GROUP BY DATE(payed_at)
)
SELECT dt, gmv, orders,
       LAG(gmv) OVER (ORDER BY dt) AS prev_gmv,
       ROUND( (gmv - LAG(gmv) OVER (ORDER BY dt))
              / LAG(gmv) OVER (ORDER BY dt) * 100, 1) AS mom_pct
FROM daily
ORDER BY dt DESC
LIMIT 30;
窗口函数最大的好处：保留明细行的同时加一列"聚合/排名"，不用 GROUP BY 把行压没。

67.2 类目 Top N 与用户排行

-- 每个类目（这里简化为按商品）销量 Top 3：ROW_NUMBER 分区排名
WITH item_sales AS (
  SELECT i.product_name, SUM(i.quantity) AS qty
  FROM order_item i JOIN orders o ON o.id = i.order_id
  WHERE o.status = 'PAYED'
  GROUP BY i.product_name
),
ranked AS (
  SELECT product_name, qty,
         ROW_NUMBER() OVER (ORDER BY qty DESC) AS rn
  FROM item_sales
)
SELECT product_name, qty, rn FROM ranked WHERE rn <= 3;

-- 用户复购：按用户统计订单数，再分桶
WITH u AS (
  SELECT user_id, COUNT(*) AS order_cnt
  FROM orders WHERE status = 'PAYED'
  GROUP BY user_id
)
SELECT CASE WHEN order_cnt = 1 THEN '新客(1单)'
            WHEN order_cnt BETWEEN 2 AND 3 THEN '复购(2-3单)'
            ELSE '忠诚(4+单)' END AS segment,
       COUNT(*) AS users
FROM u GROUP BY segment;
窗口函数报表用途
LAG / LEAD环比、同比，对同一列前后行取数
ROW_NUMBER / RANKTop N、类目排行、连续登录天数
SUM() OVER累计 GMV、累计下单金额

67.3 累计金额与分组小计

-- 累计 GMV：SUM() OVER 随时间累加
WITH daily AS (
  SELECT DATE(payed_at) dt, SUM(total_amount) gmv
  FROM orders WHERE status='PAYED' GROUP BY DATE(payed_at)
)
SELECT dt, gmv,
       SUM(gmv) OVER (ORDER BY dt) AS cumulative_gmv
FROM daily ORDER BY dt DESC LIMIT 15;
窗口函数是在"结果集"上再算一遍，数据量大时先在 CTE 里用 WHERE/索引把行数压小，再套窗口函数，别对着千万行裸算。

67.4 报表性能边界

报表查询和在线业务共用一个库时，必须给它画条红线。

-- 报表应在从库跑，且先压缩时间范围与行数
WITH daily AS (
  SELECT DATE(payed_at) dt, SUM(total_amount) gmv
  FROM orders
  WHERE payed_at >= DATE_SUB(CURDATE(), INTERVAL 90 DAY)  -- 先压数据量
    AND status='PAYED'
  GROUP BY DATE(payed_at)
)
SELECT dt, gmv, SUM(gmv) OVER (ORDER BY dt) cum FROM daily;
  • 原则报表走从库，不抢在线写；
  • 坑对千万行裸做窗口函数，会拖垮 Buffer Pool，影响在线业务。
