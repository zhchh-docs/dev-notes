18.1 统一实验库（可复现）

本章全部需求基于四张表。先把这段建表脚本在测试库跑一遍。

-- 用户表
CREATE TABLE customer (
  id         INT PRIMARY KEY AUTO_INCREMENT,
  name       VARCHAR(20) NOT NULL,
  level      VARCHAR(10) NOT NULL DEFAULT 'NORMAL',
  is_deleted TINYINT NOT NULL DEFAULT 0
);

-- 商品表
CREATE TABLE product (
  id         INT PRIMARY KEY AUTO_INCREMENT,
  name       VARCHAR(50) NOT NULL,
  category   VARCHAR(20) NOT NULL,
  price      DECIMAL(10,2) NOT NULL,
  stock      INT NOT NULL DEFAULT 0,
  is_deleted TINYINT NOT NULL DEFAULT 0
);

-- 订单主表
CREATE TABLE orders (
  id          INT PRIMARY KEY AUTO_INCREMENT,
  order_no    VARCHAR(20) NOT NULL,
  customer_id INT NOT NULL,
  amount      DECIMAL(10,2) NOT NULL,    -- 订单总额
  status      VARCHAR(10) NOT NULL,      -- UNPAID / PAID / DONE
  created_at  DATETIME NOT NULL,
  is_deleted  TINYINT NOT NULL DEFAULT 0
);

-- 订单明细表
CREATE TABLE order_item (
  id         INT PRIMARY KEY AUTO_INCREMENT,
  order_id   INT NOT NULL,
  product_id INT NOT NULL,
  quantity   INT NOT NULL,
  unit_price DECIMAL(10,2) NOT NULL
);

18.2 需求 1-3：日报基础

需求 1：每日订单量与 GMV

-- 口径：已支付订单，按日聚合
SELECT
  DATE(created_at) AS d,
  COUNT(*)          AS order_cnt,
  SUM(amount)       AS gmv
FROM orders
WHERE status = 'PAID'
  AND is_deleted = 0
GROUP BY DATE(created_at)
ORDER BY d;

需求 2：类目 GMV Top10

-- 穿过 order_item → product 拿到类目，按类目汇总
SELECT
  p.category,
  SUM(oi.quantity * oi.unit_price) AS cat_gmv,
  SUM(oi.quantity)                  AS qty
FROM orders o
INNER JOIN order_item oi ON oi.order_id = o.id
INNER JOIN product p      ON p.id = oi.product_id
WHERE o.status = 'PAID'
  AND o.is_deleted = 0
GROUP BY p.category
ORDER BY cat_gmv DESC
LIMIT 10;

需求 3：库存预警（可售库存 < 安全阈值）

SELECT id, name, category, stock
FROM product
WHERE stock <= 10          -- 安全阈值 10 件
  AND is_deleted = 0
ORDER BY stock ASC;

18.3 需求 4-6：复购与累计

需求 4：复购率（下过 ≥2 单的用户 / 下过单的用户总数）

WITH user_order_cnt AS (
  SELECT customer_id, COUNT(*) AS cnt
  FROM orders
  WHERE status = 'PAID' AND is_deleted = 0
  GROUP BY customer_id
)
SELECT
  COUNT(*)                                             AS paying_users,
  SUM(CASE WHEN cnt >= 2 THEN 1 ELSE 0 END)            AS repeat_users,
  ROUND(
    SUM(CASE WHEN cnt >= 2 THEN 1 ELSE 0 END) / COUNT(*)
  , 4) AS repeat_rate
FROM user_order_cnt;

需求 5：日客单价（GMV / 订单数）

SELECT
  DATE(created_at) AS d,
  COUNT(*)         AS order_cnt,
  SUM(amount)      AS gmv,
  ROUND(SUM(amount) / COUNT(*), 2) AS aov   -- Average Order Value
FROM orders
WHERE status = 'PAID' AND is_deleted = 0
GROUP BY DATE(created_at)
ORDER BY d;

需求 6：累计销售额（至今逐日累加）

WITH daily AS (
  SELECT DATE(created_at) AS d, SUM(amount) AS day_gmv
  FROM orders
  WHERE status = 'PAID' AND is_deleted = 0
  GROUP BY DATE(created_at)
)
SELECT
  d,
  day_gmv,
  SUM(day_gmv) OVER (ORDER BY d) AS cum_gmv
FROM daily
ORDER BY d;

18.4 需求 7-8：库存与沉睡

需求 7：每个类目销量 Top3 商品

WITH prod_sales AS (
  SELECT
    p.category,
    p.name,
    SUM(oi.quantity) AS qty
  FROM orders o
  INNER JOIN order_item oi ON oi.order_id = o.id
  INNER JOIN product p     ON p.id = oi.product_id
  WHERE o.status = 'PAID' AND o.is_deleted = 0
  GROUP BY p.category, p.name
),
ranked AS (
  SELECT
    category, name, qty,
    ROW_NUMBER() OVER (PARTITION BY category ORDER BY qty DESC, name) AS rn
  FROM prod_sales
)
SELECT category, name, qty
FROM ranked
WHERE rn <= 3
ORDER BY category, rn;

需求 8：30 天未支付的沉睡订单（催收名单）

SELECT id, order_no, customer_id, amount, created_at
FROM orders
WHERE status = 'UNPAID'
  AND is_deleted = 0
  AND created_at < CURDATE() - INTERVAL 7 DAY   -- 下单超 7 天还没付
ORDER BY created_at;

18.5 需求 9-10：连下用户

需求 9：日 GMV 环比（和前一天比）

WITH daily AS (
  SELECT DATE(created_at) AS d, SUM(amount) AS gmv
  FROM orders
  WHERE status = 'PAID' AND is_deleted = 0
  GROUP BY DATE(created_at)
)
SELECT
  d,
  gmv,
  LAG(gmv) OVER (ORDER BY d) AS prev_gmv,
  ROUND(
    (gmv - LAG(gmv) OVER (ORDER BY d))
    / NULLIF(LAG(gmv) OVER (ORDER BY d), 0)
  , 4) AS mom_rate
FROM daily
ORDER BY d;

需求 10：最近 30 天连续下过单的用户（简化口径：30 天内下单 ≥10 天）

WITH user_days AS (
  SELECT customer_id, COUNT(DISTINCT DATE(created_at)) AS active_days
  FROM orders
  WHERE status = 'PAID' AND is_deleted = 0
    AND created_at >= CURDATE() - INTERVAL 30 DAY
  GROUP BY customer_id
  HAVING active_days >= 10          -- 30 天里有 10 天以上来下单
)
SELECT c.id, c.name, u.active_days
FROM user_days u
INNER JOIN customer c ON c.id = u.customer_id
ORDER BY u.active_days DESC;
"真正连续"的高级写法：严格意义的"连续 N 天"要用到窗口函数做"日期减去行号分组"的经典技巧（gap-island 问题），本书在进阶篇的实战章节（第 80 章）会展开；本需求的"活跃天数"口径对运营召回已经够用。
报表 SQL 的通用拆解顺序：先定口径（哪些行算）→ 再定粒度（按什么分组）→ 然后选聚合函数 → 最后加窗口/排序/分页。10 条需求全是这个套路。
