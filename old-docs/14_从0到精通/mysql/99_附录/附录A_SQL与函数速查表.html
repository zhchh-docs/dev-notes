A.1 SQL 速查：DDL 数据定义

功能语法示例说明
建库CREATE DATABASE shop DEFAULT CHARSET utf8mb4;指定字符集防乱码
建表CREATE TABLE t (id BIGINT PRIMARY KEY, name VARCHAR(64)) ENGINE=InnoDB;显式 InnoDB 引擎
加列ALTER TABLE t ADD COLUMN remark VARCHAR(255);8.0.12+ 末尾加列走 INSTANT
删列ALTER TABLE t DROP COLUMN remark;大表走在线 DDL（第 60 章）
加索引ALTER TABLE t ADD INDEX idx_name(name);联合索引 ADD INDEX idx_a_b(a,b)
加唯一键ALTER TABLE t ADD UNIQUE KEY uk_phone(phone);重复值会插入失败
删表/库DROP TABLE t;不可回滚，先备份
改表名RENAME TABLE old TO new;原子操作
清空表TRUNCATE TABLE t;比 DELETE 快，重置自增

A.2 SQL 速查：DML 数据操作

功能语法示例说明
插入单行INSERT INTO t(name) VALUES('a');永远写列名清单
批量插入INSERT INTO t(name) VALUES('a'),('b');比循环单条快
冲突更新INSERT INTO t(k,v) VALUES(1,2) ON DUPLICATE KEY UPDATE v=v+1;UPSERT，依赖唯一键
更新UPDATE t SET price=price*0.9 WHERE id=1;先 SELECT 验证 WHERE
删除DELETE FROM t WHERE created_at < '2025-01-01';大表分批，避免长事务
从查询插入INSERT INTO t2 SELECT * FROM t1 WHERE id<100;数据迁移常用
IGNORE 插入INSERT IGNORE INTO t ...;撞唯一键静默跳过，慎用

WHEREWHERE status=1 AND price>100过滤行，避免隐式转换

A.4 SQL 速查：事务与锁

功能语法示例说明
开启事务START TRANSACTION;或 BEGIN
提交COMMIT;事务内变更落盘
回滚ROLLBACK;撤销未提交变更
隔离级别SET TRANSACTION ISOLATION LEVEL READ COMMITTED;RU/RC/RR/Serializable
加行锁SELECT ... FOR UPDATE;写冲突前先锁定行
读不锁SELECT ... LOCK IN SHARE MODE;8.0 中较少用
保存点SAVEPOINT sp; ROLLBACK TO sp;事务内局部回滚

A.5 SQL 速查：权限管理

功能语法示例说明
建账号CREATE USER 'app'@'10.0.%' IDENTIFIED BY 'pwd';限定来源网段
授权GRANT SELECT,INSERT ON shop.* TO 'app'@'10.0.%';最小权限
撤权REVOKE INSERT ON shop.* FROM 'app'@'10.0.%';精确回收
看权限SHOW GRANTS FOR 'app'@'10.0.%';审计常用
改密码ALTER USER 'app'@'10.0.%' IDENTIFIED BY 'newpwd';强密码策略
删账号DROP USER 'app'@'10.0.%';下线账号及时清理
角色CREATE ROLE r; GRANT r TO 'app'@'10.0.%';8.0 权限打包

A.10 系统信息查询速查

-- 版本与当前库
SELECT VERSION();
SELECT DATABASE();

-- 所有库与表
SHOW DATABASES;
SHOW TABLES;

-- 表结构
DESC orders;
SHOW CREATE TABLE orders\G

-- 状态变量与配置
SHOW VARIABLES LIKE 'innodb_buffer_pool_size';
SHOW GLOBAL STATUS LIKE 'Threads_connected';

-- 当前进程与连接
SELECT ID,USER,HOST,DB,COMMAND,TIME,INFO FROM information_schema.PROCESSLIST;

-- 慢查询 digest（8.0）
SELECT LEFT(DIGEST_TEXT,80), COUNT_STAR,
       ROUND(AVG_TIMER_WAIT/1e9,2) avg_ms
FROM performance_schema.events_statements_summary_by_digest
ORDER BY SUM_TIMER_WAIT DESC LIMIT 10;

-- 主从状态
SHOW REPLICA STATUS\G
